cmake_minimum_required(VERSION 3.22)
project(CaseZeroAndroid C CXX)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "Use the NDK toolchain with -DANDROID_ABI=arm64-v8a")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_link_options("-Wl,-z,max-page-size=16384")
get_filename_component(CZ_REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(CZ_ANDROID_DEPS "${CZ_REPO}/thirdparty/android" CACHE PATH "Locked dependency root")
set(CZ_NATIVE_OUTPUT_DIR "${CMAKE_BINARY_DIR}/jniLibs/arm64-v8a" CACHE PATH "APK native inputs")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CZ_NATIVE_OUTPUT_DIR}")
option(CZ_ALLOW_STUB_PPC "Build the explicitly non-playable diagnostics APK" OFF)
set(CZ_ANDROID_DXC_LIBRARY "" CACHE FILEPATH "Android ARM64 libdxcompiler.so")
set(XENON_ROOT "${CZ_ANDROID_DEPS}/xenon" CACHE PATH "Xenon source")
set(XENOS_ROOT "${CZ_ANDROID_DEPS}/xenos" CACHE PATH "Xenos source")
set(CZ_FFMPEG_PREFIX "${CZ_ANDROID_DEPS}/ffmpeg-install" CACHE PATH "Android LGPL ffmpeg")
foreach(dep sdl xenon xenos volk adrenotools)
    if(NOT EXISTS "${CZ_ANDROID_DEPS}/${dep}/CMakeLists.txt")
        message(FATAL_ERROR "Missing ${dep}; run tools/android/bootstrap.py first")
    endif()
endforeach()
if(NOT EXISTS "${CZ_PPC_DIR}/ppc_config.h")
    message(FATAL_ERROR "Set CZ_PPC_DIR to real generated code (or explicitly generate a CI stub)")
endif()
file(READ "${CZ_PPC_DIR}/ppc_config.h" PPC_CONFIG)
if(PPC_CONFIG MATCHES "CZ_PPC_STUB_IMAGE")
    if(NOT CZ_ALLOW_STUB_PPC)
        message(FATAL_ERROR "A game APK cannot contain stub PPC code")
    endif()
elseif(CZ_ALLOW_STUB_PPC)
    message(FATAL_ERROR "Diagnostic mode requires an explicitly marked stub tree")
endif()
if(NOT CZ_ALLOW_STUB_PPC AND NOT EXISTS "${CZ_ANDROID_DXC_LIBRARY}")
    message(FATAL_ERROR "A game APK requires Android DXC. Run tools/android/build_dxc.sh")
endif()
set(BUILD_SHARED_LIBS OFF)
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${CZ_ANDROID_DEPS}/sdl" "sdl")
# Build only the XEX loader with the TARGET compiler, not XenonRecomp's host tools.
set(THIRDPARTY_ROOT "${XENON_ROOT}/thirdparty")
add_subdirectory("${THIRDPARTY_ROOT}/fmt" "xenon-fmt")
add_subdirectory("${THIRDPARTY_ROOT}/xxHash/cmake_unofficial" "xenon-xxhash")
# Only the loader is needed. Do not compile/link Xenon's binutils-derived GPL
# disassembler, or statically absorb LGPL libmspack into the game library.
add_library(cz_lzx SHARED "${THIRDPARTY_ROOT}/libmspack/libmspack/mspack/lzxd.c")
target_include_directories(cz_lzx PUBLIC "${THIRDPARTY_ROOT}/libmspack/libmspack/mspack")
add_library(XenonUtils STATIC
    "${XENON_ROOT}/XenonUtils/xex.cpp"
    "${XENON_ROOT}/XenonUtils/image.cpp"
    "${XENON_ROOT}/XenonUtils/xdbf_wrapper.cpp"
    "${XENON_ROOT}/XenonUtils/xex_patcher.cpp"
    "${XENON_ROOT}/XenonUtils/memory_mapped_file.cpp"
    "${THIRDPARTY_ROOT}/tiny-AES-c/aes.c")
target_include_directories(XenonUtils PUBLIC "${XENON_ROOT}/XenonUtils" "${THIRDPARTY_ROOT}/simde"
    PRIVATE "${THIRDPARTY_ROOT}/tiny-AES-c" "${THIRDPARTY_ROOT}/TinySHA1")
target_compile_definitions(XenonUtils PRIVATE NOMINMAX)
target_link_libraries(XenonUtils PRIVATE cz_lzx)
add_subdirectory("${CZ_ANDROID_DEPS}/adrenotools" "adrenotools")
add_library(cz_volk STATIC "${CZ_ANDROID_DEPS}/volk/volk.c")
target_include_directories(cz_volk PUBLIC "${CZ_ANDROID_DEPS}/volk")
target_compile_definitions(cz_volk PUBLIC VK_USE_PLATFORM_ANDROID_KHR VK_NO_PROTOTYPES)
set_target_properties(cz_volk PROPERTIES C_VISIBILITY_PRESET hidden)
add_library(cz_android_vulkan STATIC vulkan_loader.cpp)
target_link_libraries(cz_android_vulkan PUBLIC cz_volk adrenotools dl)
set(CZ_SPLIT_DEBUG OFF CACHE BOOL "" FORCE)
set(CZ_BUNDLE_RPATH OFF CACHE BOOL "" FORCE)
add_subdirectory("${CZ_REPO}/runtime" "runtime")
add_library(cz_support SHARED launcher_jni.cpp "${CZ_REPO}/runtime/host/stfs_extract.cpp")
target_link_libraries(cz_support PRIVATE cz_android_vulkan log)
target_compile_definitions(cz_support PRIVATE CZ_DIAGNOSTIC=$<BOOL:${CZ_ALLOW_STUB_PPC}>)
target_link_options(cz_support PRIVATE "-Wl,--no-undefined")
# APK libraries are an explicit set, never a recursive find that can pick up host binaries.
foreach(lib cz_runtime cz_support cz_lzx SDL2 hook_impl main_hook file_redirect_hook gsl_alloc_hook)
    set_target_properties(${lib} PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CZ_NATIVE_OUTPUT_DIR}")
endforeach()
file(MAKE_DIRECTORY "${CZ_NATIVE_OUTPUT_DIR}")
file(COPY "${CZ_FFMPEG_PREFIX}/lib/libavcodec.so" "${CZ_FFMPEG_PREFIX}/lib/libavutil.so"
    DESTINATION "${CZ_NATIVE_OUTPUT_DIR}" FOLLOW_SYMLINK_CHAIN)
file(COPY "${CMAKE_SYSROOT}/usr/lib/aarch64-linux-android/libc++_shared.so"
    DESTINATION "${CZ_NATIVE_OUTPUT_DIR}")
if(CZ_ANDROID_DXC_LIBRARY)
    configure_file("${CZ_ANDROID_DXC_LIBRARY}" "${CZ_NATIVE_OUTPUT_DIR}/libdxcompiler.so" COPYONLY)
endif()
